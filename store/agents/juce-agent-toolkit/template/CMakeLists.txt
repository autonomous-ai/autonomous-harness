cmake_minimum_required(VERSION 3.22)
project(HarnessInstrument VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT JUCE_ROOT)
  message(FATAL_ERROR "Pass -DJUCE_ROOT=<installed harness>/juce")
endif()
add_subdirectory("${JUCE_ROOT}" juce EXCLUDE_FROM_ALL)
juce_add_console_app(HarnessTone PRODUCT_NAME "Harness Tone")
target_sources(HarnessTone PRIVATE Source/main.cpp)
target_compile_definitions(HarnessTone PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(HarnessTone PRIVATE juce::juce_audio_basics juce::juce_recommended_config_flags)
